def digest(path):
    with path.open('rb') as opened:
        return hashlib.file_digest(opened, 'sha256').hexdigest()


def require_stage():
    activation_path = TARGET / 'activation.json'
    if digest(activation_path) != ACTIVATION_SHA:
        raise RuntimeError('Reviewed activation artifact changed')
    activation = json.loads(activation_path.read_text())
    if activation['stage'] != str(TARGET) or activation['native_package'] != str(NATIVE):
        raise RuntimeError('Wrong target runtime/native')
    if activation['pins'] != PINS or activation['sdk'] != '0.12.1':
        raise RuntimeError('Wrong reviewed source cohort')
    if not digest(TARGET / 'staging-receipt.json') == activation['staging_receipt_sha256'] == STAGING_SHA:
        raise RuntimeError('Stage proof changed')
    if digest(PAIR_READY) != PAIR_READY_SHA:
        raise RuntimeError('Reviewed package-ready receipt changed; actual gates remain separate')
    if sys.executable != str(TARGET / 'bin/python'):
        raise RuntimeError('Run this operator with the exact target interpreter')
    verify_native_package(NATIVE)


def require_no_clients(audience):
    """Decode argv boundaries once; explicit unrelated private roots are excluded.

    No PID is signal authority here. Exact original workers are checked by the
    batch. A Toad/ACP/CLI without a distinct explicit root is not safe to exclude.
    Unknown Python processes explicitly bound to this root also refuse.
    """
    platform = Platform.current()
    for entry in Path('/proc').iterdir():
        if not entry.name.isdecimal() or int(entry.name) == os.getpid():
            continue
        try:
            if entry.stat().st_uid != os.geteuid():
                continue
            identity = ProcessIdentity.capture(int(entry.name))
            argv = tuple(os.fsdecode(item) for item in (entry / 'cmdline').read_bytes().split(b'\0') if item)
            if not argv:
                continue
            head = argv[:3]
            module = head[2] if len(head) == 3 and head[1] == '-m' else ''
            commands = {Path(arg).name for arg in head}
            client = bool(commands.intersection((*COMMANDS, 'toad-comms'))) or module in {'toad', 'agent_comms'} or module.startswith('agent_comms.')
            python = Path(head[0]).name.startswith('python')
            if not client and not python:
                continue
            environment = dict(item.split(b'=', 1) for item in (entry / 'environ').read_bytes().split(b'\0') if b'=' in item)
            selected_root = environment.get(b'AGENT_COMMS_ROOT')
            platform.require(identity)
            if selected_root is not None and os.fsdecode(selected_root) != str(ROOT):
                continue
            if module == 'agent_comms.worker' and identity in {item.process for item in audience}:
                continue  # Only these exact workers belong to the canonical retained batch.
            if client or selected_root == os.fsencode(ROOT):
                raise RuntimeError(f'Unretired or unclassified client: exact process {identity}')
        except ProcessLookupError:
            continue
        except FileNotFoundError:
            continue
        # Permission/decoding failures deliberately refuse rather than infer absence.


def require_publication_preimages(original):
    if read_active_route() != original:
        raise RuntimeError('Original route changed')
    for command in COMMANDS:
        if (LINKS / command).readlink() != CURRENT / 'bin' / command:
            raise RuntimeError('Original default link changed')
        if not (TARGET / 'bin' / command).is_file():
            raise RuntimeError('Target entrypoint missing')
        temporary = LINKS / (command + '.native-custody-publish')
        if temporary.exists() or temporary.is_symlink():
            raise RuntimeError('Prior publication temporary requires review')


@dataclass(frozen=True)
class PublishNativeCustody(StoppedOwnerInstallation):
    audience: tuple[OwnerRestartSelection, ...]
    originals: tuple  # Original typed Thread proofs, retained only in RAM.
    original_route: ActiveRoute
    route_directory: int
    receipt: Path
    viewers: frozenset[str]

    @property
    def journal_path(self):
        # Original OwnerCompactionCommit uses this exact registry sibling.
        return ROOT / "compaction-commits.sqlite3"

    def reset_paths(self):
        bus = ROOT / "bus.jsonl"
        return (ROOT / RuntimeInfoStore.filename, BusActivityIndex(bus).path,
                *(BusDisplayIndex(bus, viewer).path for viewer in sorted(self.viewers)),
                self.journal_path,
                *(self.journal_path.with_name(self.journal_path.name + suffix)
                  for suffix in ("-journal", "-wal", "-shm")))

    def require_reset_paths(self, snapshot):
        if frozenset((*snapshot.threads, *snapshot.aliases)) != self.viewers:
            raise RuntimeError("Original viewer namespace changed; recapture required")
        InputDispositions(ROOT / InputDispositions.filename).read()
        for path in (*self.reset_paths(), ROOT / InputDispositions.filename):
            try:
                info = path.lstat()
            except FileNotFoundError:
                continue
            PrivateFileRole().require(info)

    def reset_runtime(self):
        preimages = self.receipt.with_suffix(".runtime-originals")
        preimages.mkdir(mode=0o700)
        originals = []
        input_path = ROOT / InputDispositions.filename
        with InputDispositions(input_path).reading() as inputs:
            input_rows = len(inputs.rows)
            unresolved_inputs = sum(row.unresolved for row in inputs.rows.values())
            input_sha = digest(input_path) if input_path.exists() else "absent"
        for path in (*self.reset_paths(), input_path):
            try:
                with os.fdopen(os.open(path, os.O_RDONLY | os.O_NOFOLLOW), "rb") as opened:
                    before = os.fstat(opened.fileno())
                    PrivateFileRole().require(before)
                    payload = opened.read()
                    if not (FileRevision.from_stat(before) == FileRevision.from_stat(os.fstat(opened.fileno())) == FileRevision.from_stat(path.lstat())):
                        raise RuntimeError("Runtime preimage changed under stopped custody")
            except FileNotFoundError:
                continue
            captured = preimages / path.name
            descriptor = os.open(captured, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
            with os.fdopen(descriptor, "wb") as output:
                output.write(payload)
                output.flush()
                os.fsync(output.fileno())
            originals.append({"name": path.name, "sha256": hashlib.sha256(payload).hexdigest()})
        descriptor = os.open(preimages, os.O_RDONLY | os.O_DIRECTORY)
        try:
            os.fsync(descriptor)
        finally:
            os.close(descriptor)
        self.note("runtime-and-protected-input-preimages-retained", original_files=originals,
                  protected_input_rows=input_rows, protected_unresolved_inputs=unresolved_inputs,
                  protected_input_sha256=input_sha)
        RuntimeInfoStore(ROOT / RuntimeInfoStore.filename).replace({})
        for path in self.reset_paths()[1:]:
            path.unlink(missing_ok=True)
        # 00-RULES18: this journal is runtime state, not durable native proof.
        # No old source_json/schema decoding or carry. The original owner creates
        # the empty current schema only after the original file has been retained.
        CompactionJournal(self.journal_path)
        if (digest(input_path) if input_path.exists() else "absent") != input_sha:
            raise RuntimeError("Protected original input dispositions changed during reset")
        descriptor = os.open(ROOT, os.O_RDONLY | os.O_DIRECTORY)
        try:
            os.fsync(descriptor)
        finally:
            os.close(descriptor)
        self.note("only-declared-runtime-projections-reset")

    def note(self, phase, **facts):
        previous = json.loads(self.receipt.read_text())
        previous.update(phase=phase, **facts)
        _atomic_write_text(self.receipt, json.dumps(previous, indent=2) + '\n', fsync_parent=True)

    def require_selection(self, snapshot, owners):
        if {thread.name for thread in owners} != {item.name for item in self.audience}:
            raise RuntimeError('Original complete owner audience changed')
        for selection, original in zip(self.audience, self.originals, strict=True):
            current = selection.require_current(snapshot)
            current.require_idle()
            if current != original:
                raise RuntimeError('Original configuration changed; recapture/review required')
        self.require_reset_paths(snapshot)
        require_publication_preimages(self.original_route)
        require_no_clients(self.audience)

    def after_stopped(self, lifecycle):
        # Canonical batch holds wire custody and has proved every original exit.
        require_publication_preimages(self.original_route)
        require_no_clients(self.audience)
        snapshot = lifecycle.registry.snapshot()
        for original in self.originals:
            if snapshot.threads[original.name] != original:
                raise RuntimeError('Original configuration changed during retirement')
        self.require_reset_paths(snapshot)
        self.note('all-original-owners-stopped')
        self.reset_runtime()
        target_route = replace(self.original_route, native_package=NATIVE)
        _publish_active_route_locked(target_route, active_route_path(), self.route_directory,
                                     expected=self.original_route)
        self.note('target-native-route-published')
        for command in COMMANDS:
            link = LINKS / command
            if link.readlink() != CURRENT / 'bin' / command:
                raise RuntimeError('Default changed during publication; leave stopped for review')
            temporary = LINKS / (command + '.native-custody-publish')
            temporary.symlink_to(TARGET / 'bin' / command)
            temporary.replace(link)
        directory = os.open(LINKS, os.O_RDONLY | os.O_DIRECTORY)
        try:
            os.fsync(directory)
        finally:
            os.close(directory)
        if read_active_route() != target_route:
            raise RuntimeError('Target route readback differs')
        self.note('target-route-and-five-defaults-published')
        # Inherited complete() launches the SAME original credential/argument handoff.


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--execute', action='store_true')
    parser.add_argument('--receipt', type=Path, required=True)
    parser.add_argument('--joint-gate-artifact', type=Path)
    parser.add_argument('--joint-gate-sha256')
    parser.add_argument('--bus-gate-artifact', type=Path)
    parser.add_argument('--bus-gate-sha256')
    args = parser.parse_args()
    if args.receipt.exists() or args.receipt.is_symlink():
        raise RuntimeError('Existing attempt requires review, never automatic rerun')
    require_stage()
    original = read_active_route()
    if original != ActiveRoute(ROOT, ROOT_ID, OLD_NATIVE):
        raise RuntimeError('Original route/root/native differs from reviewed release')
    service = Comms(ROOT, private_initial_writes=False, private_claim_writes=False)
    snapshot = service.registry.snapshot()
    owners = tuple(thread for thread in snapshot.threads.values()
                   if thread.role.executable and snapshot.statuses[thread.name].active and thread.process_alive)
    if len(owners) != 13:
        raise RuntimeError('Expected exactly thirteen original live executable owners')
    audience = tuple(OwnerRestartSelection.capture(snapshot, thread.name) for thread in owners)
    for thread in owners:
        thread.require_idle()
        RetainedOwnerLaunch.capture(thread, snapshot, interpreter=str(SOURCE / 'bin/python'))
    directory = os.open(active_route_path().parent, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        fcntl.flock(directory, fcntl.LOCK_EX | fcntl.LOCK_NB)
        operation = PublishNativeCustody(audience, owners, original, directory, args.receipt,
                                           frozenset((*snapshot.threads, *snapshot.aliases)))
        operation.require_selection(snapshot, owners)
        print(json.dumps({'state': 'read-only-preflight', 'owners': len(owners),
                          'clients': 0, 'source_interpreter': str(SOURCE / 'bin/python')}), flush=True)
        if not args.execute:
            return
        gates = {}
        for label, artifact, expected in (
            ('joint-native-ui', args.joint_gate_artifact, args.joint_gate_sha256),
            ('bus-hot-views', args.bus_gate_artifact, args.bus_gate_sha256),
        ):
            if artifact is None or expected is None:
                raise RuntimeError('Execution requires BOTH parent-reviewed actual gate artifacts and hashes')
            if artifact == PAIR_READY or artifact == TARGET / 'activation.json' or artifact == TARGET / 'staging-receipt.json':
                raise RuntimeError('Package/activation evidence is not an actual journey gate')
            if digest(artifact) != expected:
                raise RuntimeError(f'Actual {label} gate artifact differs')
            gates[label] = {'artifact': str(artifact), 'sha256': expected}
        args.receipt.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
        if args.receipt.parent.stat().st_mode & 0o077:
            raise RuntimeError('Receipt directory must be private (0700)')
        fd = os.open(args.receipt, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
        with os.fdopen(fd, 'w') as opened:
            json.dump({'phase': 'preflight-complete', 'started': time.time(),
                       'target': str(TARGET), 'native': str(NATIVE),
                       'actual_gate_artifacts': gates,
                       'owners_before': FieldCodec.encode(audience)}, opened, indent=2)
            opened.flush()
            os.fsync(opened.fileno())
        service.owners.pin_private_nk_launch(ROOT, ROOT_ID, NATIVE)
        runtime = RestartEnvironment(path=str(TARGET / 'bin') + ':' + os.environ['PATH'],
                                     virtual_env=str(TARGET))
        results = service.owners.restart_owners(runtime=runtime,
            source_interpreter=str(SOURCE / 'bin/python'), cutover=operation)
        after = service.registry.snapshot()
        for previous, result in zip(owners, results, strict=True):
            current = after.threads[result.thread]
            if result.thread != previous.name or replace(current, process_identity=previous.process_identity) != previous:
                raise RuntimeError('Configuration readback differs; review before further actions')
            RetainedOwnerLaunch.capture(current, after, interpreter=str(TARGET / 'bin/python'))
        operation.note('retained-batch-launched-configurations-verified-ui-gate-pending',
                       finished=time.time(), results=FieldCodec.encode(results))
        print(json.dumps({'state': 'published-and-launched', 'owners': len(results)}), flush=True)
    finally:
        os.close(directory)


if __name__ == '__main__':
    main()
